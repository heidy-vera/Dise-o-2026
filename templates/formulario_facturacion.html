from flask_wtf import FlaskForm from wtforms import ( StringField, DateField, FloatField, IntegerField, SelectField, SubmitField ) from wtforms.validators import ( DataRequired, NumberRange ) class FacturacionForm(FlaskForm): numero_factura = StringField( "Número de factura", validators=[ DataRequired( message="El número de factura es obligatorio." ) ] ) cliente = SelectField( "Cliente", coerce=int, validators=[ DataRequired( message="Seleccione un cliente." ) ] ) fecha = DateField( "Fecha", validators=[ DataRequired( message="La fecha es obligatoria." ) ], format="%Y-%m-%d" ) producto = SelectField( "Producto", coerce=int, validators=[ DataRequired( message="Seleccione un producto." ) ] ) cantidad = IntegerField( "Cantidad", validators=[ DataRequired( message="La cantidad es obligatoria." ), NumberRange( min=1, message="La cantidad debe ser mayor que 0." ) ] ) # El precio se obtiene automáticamente # desde PostgreSQL. precio = FloatField( "Precio unitario" ) estado = SelectField( "Estado", choices=[ ("Pagada", "Pagada"), ("Pendiente", "Pendiente") ], validators=[ DataRequired( message="Seleccione un estado." ) ] ) submit = SubmitField( "Guardar factura" )
